// ─── PROFILES ────────────────────────────────────────────────────────────────

profile(minimal) = {
  state_store: files(JSON/JSONL),
  knowledge:   { stride: JSONL, rap: JSONL },
  opsroom:     static-html,                     // open vsf_diagram.html manually
  services:    none,                            // everything runs in-process
  deps:        [python≥3.10, pyyaml]
}

profile(standard) = {
  state_store: redis + postgresql,
  knowledge:   { rag: lancedb, embed: fastembed, rap: JSONL },
  opsroom:     fastapi(:8080) + html,
  services:    [redis, postgresql, fastapi],
  deps:        [docker|podman]
}

profile(full) = {
  state_store: redis + postgresql + neo4j,
  knowledge:   { rag: lancedb, embed: fastembed, rap: JSONL, sync_daemon: true },
  opsroom:     react(:3000) + fastapi(:8080) + websocket,
  services:    [redis, postgresql, neo4j, fastapi, nodejs],
  deps:        [docker|podman, nodejs]
}

// ─── LAYER MAP ───────────────────────────────────────────────────────────────

// Layer 1 — State Store (Cybernet implementation)
// The data channel between S1-S5. Every VSF system needs at least one.
// files → sqlite → redis → postgresql → neo4j     (complexity order)

// Layer 2 — Knowledge Layer (RAG/RAP)
// Optional in minimal, recommended in standard+.
// rag_store: reactive semantic search (LanceDB local, no server)
// rap_proposals: parameter update proposals (rap_proposals.jsonl)
// stride_history: baseline for trajectory mode (stride_history.jsonl)
// embedding_engine: vectorize documents (fastembed, no API key)

// Layer 3 — Opsroom (S5 decision space)
// static-html: vsf_diagram.html — no server, open in browser
// fastapi-html: FastAPI serves HTML with live data from state store
// react-app:    React SPA + WebSocket for real-time updates

// ─── PER-COMPONENT DEPENDENCIES ──────────────────────────────────────────────

S1.any           → python≥3.10
S2.coordinator   → state_store ∈ {sqlite, redis}
S3.control       → state_store ∈ {sqlite, redis}
S3*.emergency    → state_store ∈ {files, sqlite}
S3*.trajectory   → stride_history(N≥10)
S4.checo         → rag_store + run_history(N≥5)
S5.opsroom       → fastapi + state_store
knowledge.rag    → lancedb + fastembed
knowledge.rap    → JSONL + opsroom_ui(approval_flow)
